# Thermal Engine Studio - Python Dependencies

# Qt GUI framework (multiplataforma: Windows, Linux, macOS)
PySide6>=6.5.0

# Image processing
Pillow>=10.0.0

# System monitoring (en Linux también lee CPU temp/frecuencia y RAPL)
psutil>=5.9.0

# Servidor web local (destino Web: /image.jpg + panel de control)
Flask>=3.0.0

# HID device communication (for LCD display)
# En Linux usa el backend hidraw/libusb; ver reglas udev en scripts/.
hidapi>=0.14.0

# Comunicación USB bulk directa (protocolo LY) para el LCD Thermalright
# Trofeo Vision (0416:5408). Necesario en Linux; en Windows el dispositivo
# se expone por WinUSB y pyusb necesita una DLL de libusb-1.0, que aporta
# 'libusb-package' (ver más abajo).
pyusb>=1.2.1

# Backend libusb-1.0 para pyusb en Windows (incluye la DLL). Sin esto,
# usb.core.find() lanza "No backend available" y el LCD no se detecta.
libusb-package>=1.0.26 ; sys_platform == "win32"

# Video background support (optional)
opencv-python>=4.8.0
numpy>=1.24.0

# --- Sensores GPU NVIDIA (opcional en Linux y Windows) ---------------------
# Lectura de la GPU vía NVML. Si no está instalado se recurre a 'nvidia-smi'.
# En Windows el backend nativo lo usa sin privilegios de administrador.
nvidia-ml-py>=12.0.0 ; sys_platform == "linux" or sys_platform == "win32"

# --- Sensores de CPU en Windows (opcional) ---------------------------------
# 'WMI' (que instala pywin32) permite leer la frecuencia y la temperatura de
# CPU vía ACPI. El backend nativo lo carga en segundo plano si falta.
WMI>=1.5.1 ; sys_platform == "win32"
